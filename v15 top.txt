local function consumeSpread(st, S, isADS, t)
local base = S.HipSpreadDegrees or 0
local ads = S.ADSSpreadDegrees or base
if isADS then
base = ads or base
end
local rec = S.SpreadRecoveryPerSec or 0
local since
if st.lastT > 0 then
since = math.max(0, t - st.lastT)
else
since = math.huge
end
if rec > 0 then
base = math.max(base, st.cur - rec * since)
end
local maxS = S.SpreadMaxDegrees or base
local hipInc = S.HipSpreadIncreasePerShot
local adsInc = S.ADSSpreadIncreasePerShot
local inc = (type(hipInc) ~= "number") and (S.SpreadIncreasePerShot or 0) or hipInc
if type(adsInc) ~= "number" then
adsInc = S.SpreadIncreasePerShot or inc
end
if isADS then
inc = adsInc or inc
end
st.cur = math.min(maxS, base + inc)
st.lastT = t
return base
end
local function applySpread(dir, seed, seq, deg)
if type(deg) ~= "number" or deg <= 0 then
return dir
end
local tn = math.tan(math.rad(deg))
local s = math.floor(tonumber(seed) or 1)
local sq = math.floor(tonumber(seq) or 0)
local k = (s + sq * 48271 + 12820163) % 2147483647
local rng = Random.new(k <= 0 and 1 or k)
local r1 = rng:NextNumber()
local r2 = rng:NextNumber()
local r = math.sqrt(r1) * tn
local th = r2 * math.pi * 2
local d = dir.Unit
local up = Vector3.new(0, 1, 0)
local right = d:Cross(math.abs(d:Dot(up)) > 0.95 and Vector3.new(1, 0, 0) or up).Unit
local upv = right:Cross(d).Unit
return (d + right * (math.cos(th) * r) + upv * (math.sin(th) * r)).Unit
end
local function compensate(target, seed, seq, deg)
if deg <= 0 then
return target
end
local d = target
for _ = 1, 10 do
local a = applySpread(d, seed, seq, deg)
d = (d + (target - a)).Unit
end
if d ~= d then
return target
end
return d
end

local function solveAim(origin, part, S, usePredict)
local pos = part.Position
if not usePredict then
return (pos - origin).Unit
end
local model = part:FindFirstAncestorOfClass("Model")
local root = model and model:FindFirstChild("HumanoidRootPart")
local vel = (root and root.AssemblyLinearVelocity) or part.AssemblyLinearVelocity
local speed = S and S.BulletSpeed
if type(speed) ~= "number" or speed <= 0 then
return (pos - origin).Unit
end
local t = (pos - origin).Magnitude / speed
local aim = pos
for _ = 1, 5 do
aim = pos + vel * t
t = (aim - origin).Magnitude / speed
end
if S and S.BulletDrop == true then
local g = Workspace.Gravity * ((type(S.BulletGravityMult) == "number") and S.BulletGravityMult or 1)
local sd = S.BulletDropStartDistance
if type(sd) ~= "number" then
local fr = S.BulletDropStartFraction
fr = (type(fr) == "number") and fr or 0
if fr > 1 then
fr = fr / 100
end
sd = (S.MaxDistance or C.SADist or 600) * math.clamp(fr, 0, 1)
end
local startT = sd / math.max(speed, 1)
local td = math.max(0, t - startT)
aim = aim + Vector3.new(0, 0.5 * g * td * td, 0)
end
return (aim - origin).Unit
end

local function isGunArgs(a)

end
local function processShot(...)
local a = table.pack(...)
if not isGunArgs(a) then
return nil
end
lastShotArgs = table.clone(a)
lastShotToolName = a[4]
local char = LP.Character
local tool = char and char:FindFirstChild(a[4])
local S = tool and getSettings(tool)
local origin = a[2]
local muzzle = a[1]
local maxDist = (S and S.MaxDistance) or C.SADist or 600
local wanted = a[3]
local fromSA = false
if C.SA then
local part = getTarget(origin)
if C.MuzzleOffset and part then
local newMuzzle = getBestMuzzle(part.Position)
muzzle = newMuzzle
origin = newMuzzle
a[1] = newMuzzle
a[2] = newMuzzle
part = getTarget(origin) or part
end
if part then
wanted = solveAim(origin, part, S, C.Pr)
fromSA = true
end
end
if wanted == a[3] and C.AutoFire then
local part = getAutoFireTarget(origin)
if part then
wanted = solveAim(origin, part, S, C.AFPr)
end
end
local sendDir = wanted
local realDir = wanted
if S and tool then
local st = spreadState[tool]
if not st then
st = { cur = 0, lastT = 0 }
spreadState[tool] = st
end
local deg = consumeSpread(st, S, a[6] == true, a[5] or Workspace:GetServerTimeNow())
local seed = tool:GetAttribute("GunSpreadSeed")
if deg > 0 and seed ~= nil then
if C.NoSpread then
sendDir = compensate(wanted, seed, a[8], deg)
realDir = wanted
else
realDir = applySpread(wanted, seed, a[8], deg)
end
end
end
queueLine(muzzle, origin, realDir, maxDist)
if sendDir ~= a[3] then
a[3] = sendDir
return a
end
return a
end

args.n = math.max(args.n or 8, 8)

tool:SetAttribute("AstraAutoSeq", seq)

args = table.pack(muzzle, muzzle, dir, tool.Name, now, false, session, seq)
end
if S then
local st = spreadState[tool]

st = { cur = 0, lastT = 0 }
spreadState[tool] = st
end
local deg = consumeSpread(st, S, false, now)
local seed = tool:GetAttribute("GunSpreadSeed")
if deg > 0 and seed ~= nil and C.NoSpread then
args[3] = compensate(dir, seed, args[8], deg)
end
end
queueLine(muzzle, muzzle, dir, (S and S.MaxDistance) or C.SADist)

remote:FireServer(table.unpack(args, 1, args.n or #args))

local function trackHumanoid(hum)

return
end
lastHealth[hum] = hum.Health
pendingDamage[hum] = 0
hum.HealthChanged:Connect(function(new)
local prev = lastHealth[hum]

pendingDamage[hum] = (pendingDamage[hum] or 0) + (prev - new)

lastHealth[hum] = new
end)
end

local TELEPORT_UP = 2
local ACCEL = 0.4
local DECEL = 0.4
local TOTAL = 2.5
local RESTART_BLOCK = 2.4
local SLIDE_ANTIDETECT_TIME = 1.5
local SLIDE_ANTIDETECT_SPEED = 1
local isSliding = false
local slideConn, bodyVel, bodyPos, slideThread
local startY, blockUntil, slideStartT = 0, 0, 0
local function getCharEl()
local ch = LP.Character
if not ch then
return nil, nil
end
return ch:FindFirstChild("Humanoid"), ch:FindFirstChild("HumanoidRootPart")
end
local function stopSlide()
isSliding = false
blockUntil = tick() + RESTART_BLOCK
if slideThread then
pcall(task.cancel, slideThread)
slideThread = nil
end
if slideConn then
slideConn:Disconnect()
slideConn = nil
end
if bodyVel then
bodyVel:Destroy()
bodyVel = nil
end
if bodyPos then
bodyPos:Destroy()
bodyPos = nil
end
local h = getCharEl()
if h then
h:ChangeState(Enum.HumanoidStateType.Freefall)
h.WalkSpeed = 16
end
end
local function startSlide()
if not C.SlideOn or tick() < blockUntil or isSliding then
return
end
local h, rt = getCharEl()
if not h or not rt then
return
end
isSliding = true
slideStartT = tick()
startY = rt.Position.Y
if C.SlideTP then
rt.CFrame = rt.CFrame * CFrame.new(0, TELEPORT_UP, 0)
startY = rt.Position.Y
end
bodyPos = Instance.new("BodyPosition")
bodyPos.MaxForce = Vector3.new(0, math.huge, 0)
bodyPos.Position = rt.Position
bodyPos.Parent = rt
bodyVel = Instance.new("BodyVelocity")
bodyVel.MaxForce = Vector3.new(math.huge, 0, math.huge)
bodyVel.Velocity = Vector3.new(0, 0, 0)
bodyVel.Parent = rt
local maxS = C.SlideSpeed
slideConn = RunService.Heartbeat:Connect(function()
local h2, rt2 = getCharEl()
if not (h2 and rt2 and bodyVel and bodyPos) then
stopSlide()
return
end
local elapsed = tick() - slideStartT
local yP = math.clamp(elapsed / ACCEL, 0, 1)
local targetY = startY + (C.SlideTP and TELEPORT_UP or 0) * yP
if C.AntiDetect and elapsed > SLIDE_ANTIDETECT_TIME then
local descendAmount = (elapsed - SLIDE_ANTIDETECT_TIME) * SLIDE_ANTIDETECT_SPEED
targetY = targetY - descendAmount
end
bodyPos.Position = Vector3.new(rt2.Position.X, targetY, rt2.Position.Z)
local spd = 0
if elapsed < ACCEL then
spd = maxS * (elapsed / ACCEL)
elseif elapsed < TOTAL - DECEL then
spd = maxS
else
spd = maxS * (1 - (elapsed - (TOTAL - DECEL)) / DECEL)
end
local md = h2.MoveDirection
bodyVel.Velocity = (md.Magnitude > 0 and md or rt2.CFrame.LookVector) * spd
h2.WalkSpeed = 16
if elapsed >= TOTAL then
stopSlide()
end
end)
slideThread = task.delay(TOTAL, stopSlide)
end

local function serializeC()
local data = {}
for k, v in pairs(C) do
if typeof(v) == "Color3" then
data[k] = { __color = true, r = v.R, g = v.G, b = v.B }
elseif type(v) ~= "table" then
data[k] = v
end
end
return HttpService:JSONEncode(data)
end
local function deserializeC(json)
local ok, data = pcall(function()
return HttpService:JSONDecode(json)
end)
if not ok or type(data) ~= "table" then
return false
end
for k, v in pairs(data) do
if type(v) == "table" and v.__color then
C[k] = Color3.new(v.r, v.g, v.b)
else
C[k] = v
end
end
return true
end
